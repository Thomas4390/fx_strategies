\section{Performance de recherche in-sample}
\label{sec:is_results}

\begin{warningbox}
\textbf{Cette section mesure le moteur de recherche, pas le compte.} Les chiffres qui suivent proviennent du backtest \texttt{vectorbtpro}~; ils servent à éprouver la structure de la stratégie sur des trajectoires alternatives, pas à annoncer une performance. Les chiffres de performance publiés sont ceux de l'exécution MetaTrader~5 (section~\ref{sec:executive_summary}). Les deux ne sont pas transposables l'un à l'autre --- la section~\ref{sec:research_vs_execution} explique pourquoi.
\end{warningbox}

\begin{insightbox}
\textbf{Un repli de 55\,\% sur la mesure de recherche.} Ce n'est pas une anomalie de calcul~: c'est ce que produit mécaniquement une cible de volatilité de 37\,\% appliquée à un portefeuille dont la volatilité naturelle est de quelques points. Le moteur de recherche atteint cette cible~; le moteur d'exécution, dont le dimensionnement passe par un budget de risque borné par le stop, n'y parvient pas et réalise un repli plus faible. L'écart entre les deux chiffres est une différence de définition, pas une contradiction.
\end{insightbox}

\input{tables/walkforward_yearly.tex}

\begin{itemize}
\item \textbf{2019 ($+1.22$)}~: le régime de fin de cycle, longtemps présenté comme défavorable à la stratégie, lui est en réalité favorable sous la configuration actuelle.
\item \textbf{2020 ($+0.88$)}~: année du choc Covid, positive malgré tout. Le filtre macro bloque le Moteur~1 pendant l'essentiel de la période, le privant du rebond.
\item \textbf{2021 ($+0.45$)}~: la fenêtre la plus terne des positives.
\item \textbf{2022 ($+2.31$)}~: le cycle de hausses de la Fed et la parité EUR-USD alimentent le suivi de tendance.
\item \textbf{2023 ($-0.20$)}~: seule fenêtre négative de la série. C'est aussi la seule année négative du compte à l'exécution.
\item \textbf{2024 ($+2.21$)}~: seconde meilleure fenêtre, les quatre moteurs alignés.
\item \textbf{2025 à avril 2026 ($+3.45$)}~: dernière fenêtre, plus longue que les autres. Elle recouvre en partie la tranche de données gelée par la politique de \emph{holdout}~--- elle est rapportée ici à titre de mesure, elle n'a servi à aucun choix de paramètre.
\end{itemize}

\begin{warningbox}
\textbf{Ces Sharpe de fenêtre ne sont pas des rendements annuels.} Ils mesurent la régularité du signal à l'intérieur de chaque fenêtre, pas ce que le compte aurait gagné. Le résultat année par année du compte figure au tableau~\ref{tab:mt5_yearly}, et il raconte une histoire voisine~: une seule année négative, 2023, et une concentration du gain sur 2024--2026.
\end{warningbox}

\subsection{Corrélations glissantes entre moteurs}

\begin{figure}[htbp]
\centering
\includegraphics[width=\linewidth]{rolling_correlation_63d.png}
\caption{Corrélation glissante à 63~jours pour les six paires de moteurs, moyenne de période en légende. En haut, les paires internes au trio FX~: Momentum Journalier $\times$ RSI Journalier reste majoritairement négative ($-0{,}24$ en moyenne), ce qui est le levier de diversification identifié sur la pleine période. En bas, les trois paires impliquant la sleeve momentum~: les moyennes sont faibles ($-0{,}00$, $+0{,}18$, $-0{,}08$), ce qui fonde la contribution de ce moteur à la diversification. \textbf{La moyenne masque toutefois des épisodes marqués}~: la corrélation entre la sleeve momentum et le Momentum Journalier atteint $+0{,}65$ sur plusieurs séquences de quelques mois --- les deux moteurs traitent d'ailleurs le même dollar-yen depuis juillet~2026 --- deux moteurs de suivi de tendance s'alignent naturellement lorsque les régimes sont directionnels. La décorrélation est une propriété de long terme, pas une garantie instantanée~: sur ces épisodes, les deux moteurs perdent ou gagnent ensemble.}
\end{figure}
